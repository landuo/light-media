    @staticmethod
    def is_magnet_link(content: Union[str, bytes]) -> bool:
        """
        判断内容是否为磁力链接
        """
        if not content:
            return False
        if isinstance(content, str) and content.startswith("magnet:"):
            return True
        if isinstance(content, bytes) and content.startswith(b"magnet:"):
            return True
        return False
